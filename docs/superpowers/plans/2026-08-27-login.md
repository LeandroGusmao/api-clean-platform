# Fatia 3 — Login: plano de implementação

> **Para executores agênticos:** SUB-SKILL OBRIGATÓRIA: use
> `superpowers:subagent-driven-development` (recomendado) ou
> `superpowers:executing-plans` para implementar tarefa a tarefa. Os passos usam
> checkbox (`- [ ]`) para acompanhamento.

**Goal:** entregar `POST /auth/login`, um guard de Bearer genérico e
`GET /auth/me`, executando a metade *stateless* da decisão 11.

**Architecture:** o use case de login compara a senha por `IHashProvider` e emite
um access JWT por `ITokenSigner`, ambos contratos na camada pura com adapters em
`infra/`. O guard é infra **global** e genérica: recebe a função de verificação
injetada, então nenhum módulo importa `auth` para se proteger. O principal viaja
num `WeakMap` não exportado, nunca em `declare module 'fastify'`.

**Tech Stack:** TypeScript 7 · Fastify 5 · Zod 4 · Drizzle 1.0.0-rc.4 ·
`@node-rs/argon2` · **`jose` 6.2.3 (nova)** · Vitest 4 · Biome.

**Spec:** [`docs/superpowers/specs/2026-08-27-login-design.md`](../specs/2026-08-27-login-design.md)

```ts
import { z } from 'zod'

/** 1 hour — decision 11. Short by design: the refresh token carries the session. */
export const DEFAULT_ACCESS_TOKEN_TTL_SECONDS = 3600

/**
 * 256 bits of secret for HS256. Verified: `jose` signs with a 5-byte secret and
 * enforces no minimum of its own, so this line is the only entropy guard in the
 * system — not a redundancy over a library check.
 */
const MIN_JWT_SECRET_LENGTH = 32

/**
 * The module owns its own variables (decision 14). `infra/config/env.ts` merges
 * this into the core schema so the process dies at boot listing every problem
 * at once, instead of revealing one per restart.
 */
export const authEnvSchema = z.object({
  JWT_SECRET: z.string().min(MIN_JWT_SECRET_LENGTH),
  // `.int().positive()` is not decoration: bare `z.coerce.number()` turns an
  // empty value into 0 in silence, and a TTL of zero mints expired tokens.
  ACCESS_TOKEN_TTL_SECONDS: z.coerce
    .number()
    .int()
    .positive()
    .default(DEFAULT_ACCESS_TOKEN_TTL_SECONDS),
})

export type AuthConfig = z.infer<typeof authEnvSchema>
```

```ts
/** Claims the API reads back from a verified access token. */
export interface AccessTokenPayload {
  /** Subject: the authenticated user's id. */
  sub: string
}

/**
 * The stateless access token (decision 11). It does NOT produce the refresh
 * token: that one is opaque, persisted and revocable, and confusing the two is
 * how a design degenerates into the discarded stateless-only option.
 *
 * Minimal claims — `sub`, `iat`, `exp`. No `jti`, which only serves a denylist
 * that decision 11 refused: a contract is born with what has a consumer.
 */
export interface ITokenSigner {
  sign(payload: AccessTokenPayload): Promise<string>
  /** Throws on any invalid, expired, forged or malformed token. */
  verify(token: string): Promise<AccessTokenPayload>
}
```

```ts
import { describe, expect, it } from 'vitest'
import type { AuthConfig } from '#modules/auth/config'
import { JwtTokenSigner } from '#modules/auth/infra/providers/jwt-token-signer'
import type { IClock } from '#shared/providers/clock'

const SECRET = 'a-test-secret-with-at-least-32-chars'
const OTHER_SECRET = 'a-different-secret-with-32-chars-ok'
const TTL_SECONDS = 3600
const USER_ID = '3f2a1b4c-5d6e-4f70-8192-a3b4c5d6e7f8'
const FIXED_NOW = new Date('2026-08-27T12:00:00.000Z')
const MILLISECONDS_PER_SECOND = 1000

/**
 * A one-method port becomes an object literal in the spec, per the convention
 * from the register slice: a fake file would land in `dist/`, since
 * `tsconfig.build.json` only excludes `*.spec.ts`.
 */
function makeClock(start: Date) {
  let current = start
  const clock: IClock = { now: () => new Date(current.getTime()) }

  return {
    clock,
    advanceSeconds(seconds: number) {
      current = new Date(current.getTime() + seconds * MILLISECONDS_PER_SECOND)
    },
  }
}

function makeConfig(secret = SECRET): AuthConfig {
  return { JWT_SECRET: secret, ACCESS_TOKEN_TTL_SECONDS: TTL_SECONDS }
}

describe('JwtTokenSigner', () => {
  it('verifies a token it just signed', async () => {
    const { clock } = makeClock(FIXED_NOW)
    const signer = new JwtTokenSigner(makeConfig(), clock)

    const token = await signer.sign({ sub: USER_ID })

    await expect(signer.verify(token)).resolves.toEqual({ sub: USER_ID })
  })

  it('accepts a token one second before it expires', async () => {
    const { clock, advanceSeconds } = makeClock(FIXED_NOW)
    const signer = new JwtTokenSigner(makeConfig(), clock)
    const token = await signer.sign({ sub: USER_ID })

    advanceSeconds(TTL_SECONDS - 1)

    await expect(signer.verify(token)).resolves.toEqual({ sub: USER_ID })
  })

  it('rejects a token past its expiry', async () => {
    // The clock reaches `jwtVerify` through `currentDate`. Without that option
    // jose would check `exp` against its own `Date.now()`, and this test would
    // need a real one-hour wait.
    const { clock, advanceSeconds } = makeClock(FIXED_NOW)
    const signer = new JwtTokenSigner(makeConfig(), clock)
    const token = await signer.sign({ sub: USER_ID })

    advanceSeconds(TTL_SECONDS + 1)

    await expect(signer.verify(token)).rejects.toThrow()
  })

  it('rejects a token signed with a different secret', async () => {
    const { clock } = makeClock(FIXED_NOW)
    const token = await new JwtTokenSigner(makeConfig(OTHER_SECRET), clock).sign({
      sub: USER_ID,
    })

    await expect(new JwtTokenSigner(makeConfig(), clock).verify(token)).rejects.toThrow()
  })

  it('rejects a malformed token', async () => {
    const { clock } = makeClock(FIXED_NOW)
    const signer = new JwtTokenSigner(makeConfig(), clock)

    await expect(signer.verify('not-a-jwt')).rejects.toThrow()
  })
})
```

```ts
import { SignJWT, jwtVerify } from 'jose'
import type { AuthConfig } from '#modules/auth/config'
import type {
  AccessTokenPayload,
  ITokenSigner,
} from '#modules/auth/providers/token-signer'
import type { IClock } from '#shared/providers/clock'

const ALGORITHM = 'HS256'
const MILLISECONDS_PER_SECOND = 1000
const MISSING_SUBJECT_MESSAGE = 'Access token is missing a subject'

/**
 * HS256 through `jose`, whose only dependency is none and which is ESM-native.
 *
 * `@fastify/jwt` was not an option: it would tie token issuance to Fastify, and
 * decision 10 requires the dependency graph to be built without an HTTP
 * framework so that a worker and a cron can share it.
 */
export class JwtTokenSigner implements ITokenSigner {
  readonly #secret: Uint8Array
  readonly #ttlSeconds: number
  readonly #clock: IClock

  constructor(config: AuthConfig, clock: IClock) {
    this.#secret = new TextEncoder().encode(config.JWT_SECRET)
    this.#ttlSeconds = config.ACCESS_TOKEN_TTL_SECONDS
    this.#clock = clock
  }

  async sign(payload: AccessTokenPayload): Promise<string> {
    const issuedAt = Math.floor(this.#clock.now().getTime() / MILLISECONDS_PER_SECOND)

    return new SignJWT({})
      .setProtectedHeader({ alg: ALGORITHM })
      .setSubject(payload.sub)
      .setIssuedAt(issuedAt)
      .setExpirationTime(issuedAt + this.#ttlSeconds)
      .sign(this.#secret)
  }

  async verify(token: string): Promise<AccessTokenPayload> {
    const { payload } = await jwtVerify(token, this.#secret, {
      // Load-bearing, not decoration: verified that a token signed HS512 with
      // this same secret verifies WITHOUT the pin.
      algorithms: [ALGORITHM],
      // Without this, jose checks `exp` against its own clock and the injected
      // one would only affect signing.
      currentDate: this.#clock.now(),
    })

    if (typeof payload.sub !== 'string') throw new Error(MISSING_SUBJECT_MESSAGE)

    return { sub: payload.sub }
  }
}
```

```ts
export interface LoginUserResult {
  accessToken: string
}

/**
 * Authenticates by email and password, issuing a short-lived access token.
 *
 * Issues no refresh token: that one is opaque, persisted and rotating, and it
 * arrives with the slice that can redeem it. Does not check `emailVerifiedAt`
 * either — nothing can verify an email yet, so the gate would lock every
 * account out. It arrives with the verification path that opens it.
 */
export class LoginUserUseCase {
  constructor(
    private readonly usersRepository: IUsersRepository,
    private readonly hashProvider: IHashProvider,
    private readonly tokenSigner: ITokenSigner,
  ) {}

  async execute(command: LoginUserCommand): Promise<LoginUserResult> {
    const email = Email.create(command.email)
    const user = await this.usersRepository.findByEmail(email)

    // Always compares, decoy included, so a missing account and a wrong
    // password cost the same. Returning early here would let the response time
    // answer "does this account exist?".
    const hashed = user?.passwordHash ?? this.hashProvider.dummyHash
    const matches = await this.hashProvider.compare(command.password, hashed)

    // One `throw`, not two branches: separate branches start identical and
    // drift apart at the first maintenance — one gains a log, the other a
    // message — and the oracle comes back.
    if (user === null || !matches) throw new InvalidCredentialsError()

    return { accessToken: await this.tokenSigner.sign({ sub: user.id }) }
  }
}
```

```ts
import type { FastifyRequest, preHandlerHookHandler } from 'fastify'
import { UnauthorizedError } from '#shared/errors/app-error'

const BEARER_PREFIX = 'Bearer '
const INVALID_ACCESS_TOKEN_CODE = 'INVALID_ACCESS_TOKEN'
const INVALID_ACCESS_TOKEN_MESSAGE = 'Invalid or expired access token'
const UNGUARDED_ROUTE_MESSAGE =
  'getPrincipal called on a route registered without the auth guard'

/** The authenticated caller, as the guard learned it from the token. */
export interface Principal {
  id: string
}

/**
 * Injected, which is the whole point: this file knows nothing about the auth
 * module, so no module has to import `auth` in order to protect its routes.
 */
export type VerifyAccessToken = (token: string) => Promise<{ sub: string }>

/**
 * Extends `UnauthorizedError`, not `AppError` directly: the error handler
 * branches on the six categories and treats an `AppError` outside them as a
 * programming mistake, answering 500.
 */
export class InvalidAccessTokenError extends UnauthorizedError {
  readonly code = INVALID_ACCESS_TOKEN_CODE

  constructor() {
    super(INVALID_ACCESS_TOKEN_MESSAGE)
  }
}

/**
 * The guard is the only writer in the system: this map is not exported, so
 * forging a principal would require importing something that does not exist.
 *
 * The alternative — `declare module 'fastify'` with `request.user` — is an
 * ambient global type that would let any handler of any module assign
 * `request.user = { id: request.body.userId }` and still compile, reached by
 * property access rather than by import, where no boundary lint can see it.
 */
const principals = new WeakMap<FastifyRequest, Principal>()

export function makeAuthGuard(verify: VerifyAccessToken): preHandlerHookHandler {
  return async (request) => {
    const header = request.headers.authorization

    if (header === undefined || !header.startsWith(BEARER_PREFIX)) {
      throw new InvalidAccessTokenError()
    }

    const token = header.slice(BEARER_PREFIX.length)

    let subject: string
    try {
      ;({ sub: subject } = await verify(token))
    } catch {
      // Swallowed on purpose: the library's message must not reach the client,
      // and a missing header, a forged token and an expired one are one single
      // state to a client that knows what it sent.
      throw new InvalidAccessTokenError()
    }

    principals.set(request, { id: subject })
  }
}

/**
 * Only reachable when a route was registered without the guard — a wiring
 * mistake, not a credential failure. A plain `Error` on purpose: it lands in the
 * error handler's generic branch, so the mistake is LOGGED as a 500 with its
 * traceId instead of answering 401 and reading like ordinary expired-token
 * traffic. The client still sees only the generic body.
 */
export function getPrincipal(request: FastifyRequest): Principal {
  const principal = principals.get(request)
  if (principal === undefined) throw new Error(UNGUARDED_ROUTE_MESSAGE)

  return principal
}
```

```ts
import type { FastifyPluginAsync } from 'fastify'
import type { ZodTypeProvider } from 'fastify-type-provider-zod'
import { z } from 'zod'
import { errorResponseSchema } from '#infra/http/error-response'
import type { AuthContainer } from '#modules/auth/container'
import { MAX_EMAIL_LENGTH } from '#modules/auth/value-objects/email'
import { MAX_PASSWORD_LENGTH } from '#modules/auth/value-objects/password'
import {
  AUTH_TAG,
  BAD_REQUEST_STATUS,
  OK_STATUS,
  UNAUTHORIZED_STATUS,
} from '#modules/auth/infra/http/constants'

const LOGIN_PATH = '/login'
const LOGIN_SUMMARY = 'Log in'
const LOGIN_DESCRIPTION =
  'Exchanges an email and a password for a short-lived access token, to be ' +
  'sent as `Authorization: Bearer <token>` on protected routes.'

const MIN_SUBMITTED_PASSWORD_LENGTH = 1

/**
 * Presence, not policy. Enforcing the registration rules here would answer 400
 * where the correct answer is 401, telling an attacker the guess failed the
 * RULES rather than the password — and an account created under a looser policy
 * must stay able to log in.
 *
 * The maximum stays for an unrelated reason: without a ceiling, a 10 MB body
 * becomes argon2 over 10 MB.
 */
const bodySchema = z.object({
  email: z.string().trim().max(MAX_EMAIL_LENGTH).pipe(z.email()),
  password: z.string().min(MIN_SUBMITTED_PASSWORD_LENGTH).max(MAX_PASSWORD_LENGTH),
})

const okSchema = z.object({ accessToken: z.string() })

export const loginRoute: FastifyPluginAsync<{ container: AuthContainer }> = async (
  app,
  { container },
) => {
  app.withTypeProvider<ZodTypeProvider>().post(
    LOGIN_PATH,
    {
      schema: {
        tags: [AUTH_TAG],
        summary: LOGIN_SUMMARY,
        description: LOGIN_DESCRIPTION,
        body: bodySchema,
        response: {
          [OK_STATUS]: okSchema,
          [BAD_REQUEST_STATUS]: errorResponseSchema,
          [UNAUTHORIZED_STATUS]: errorResponseSchema,
        },
      },
    },
    async (request, reply) => {
      const { accessToken } = await container.loginUser.execute(request.body)

      return reply.status(OK_STATUS).send({ accessToken })
    },
  )
}
```

```ts
import type { FastifyPluginAsync, preHandlerHookHandler } from 'fastify'
import type { ZodTypeProvider } from 'fastify-type-provider-zod'
import { z } from 'zod'
import { getPrincipal } from '#infra/http/auth-guard'
import { errorResponseSchema } from '#infra/http/error-response'
import type { AuthContainer } from '#modules/auth/container'
import {
  AUTH_TAG,
  NOT_FOUND_STATUS,
  OK_STATUS,
  UNAUTHORIZED_STATUS,
} from '#modules/auth/infra/http/constants'

const ME_PATH = '/me'
const ME_SUMMARY = 'Get the authenticated user profile'
const ME_DESCRIPTION =
  'Returns the profile of the user the access token was issued for. The id ' +
  'always comes from the token subject, never from client input.'

const BEARER_SECURITY_SCHEME = 'bearerAuth'

/**
 * `passwordHash` and `updatedAt` are out: the first for the obvious reason, the
 * second for having no consumer. `emailVerifiedAt` is a reserved seat — always
 * null until the verification slice — and it is published now so the client's
 * shape does not change then.
 */
const userSchema = z.object({
  id: z.uuid(),
  email: z.email(),
  emailVerifiedAt: z.iso.datetime().nullable(),
  createdAt: z.iso.datetime(),
})

export const meRoute: FastifyPluginAsync<{
  container: AuthContainer
  guard: preHandlerHookHandler
}> = async (app, { container, guard }) => {
  app.withTypeProvider<ZodTypeProvider>().get(
    ME_PATH,
    {
      preHandler: guard,
      schema: {
        tags: [AUTH_TAG],
        summary: ME_SUMMARY,
        description: ME_DESCRIPTION,
        security: [{ [BEARER_SECURITY_SCHEME]: [] }],
        response: {
          [OK_STATUS]: userSchema,
          [UNAUTHORIZED_STATUS]: errorResponseSchema,
          [NOT_FOUND_STATUS]: errorResponseSchema,
        },
      },
    },
    async (request, reply) => {
      const user = await container.getUserProfile.execute(getPrincipal(request).id)

      // The dates are serialized by hand: `z.iso.datetime()` rejects a `Date`,
      // and because declaring a status makes the serializer validate that
      // status's body, returning the entity's Date answers 500, not 200.
      return reply.status(OK_STATUS).send({
        id: user.id,
        email: user.email.value,
        emailVerifiedAt: user.emailVerifiedAt?.toISOString() ?? null,
        createdAt: user.createdAt.toISOString(),
      })
    },
  )
}
```
